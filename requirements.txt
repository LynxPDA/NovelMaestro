# NovelMaestro: зависимости рантайма (pip install -r requirements.txt)
#
# Политика (AGENTS.md §2): внешних библиотек минимум. Каждая обязана или закрывать
# роль целиком, или иметь явный фолбэк; установка без интернета и на экзотической
# платформе не должна падать. Активный стек печатает `python3 -m core.deps`
# (и он же — строкой в логе web-сервера при старте).
#
# Разработка (тесты) — requirements-dev.txt, он включает этот файл.

# HTTP-транспорт LLM — единственный. Один клиент на процесс переиспользует
# соединения (конвейер бьёт сотнями запросов в один хост) и даёт раздельные
# connect/write/read таймауты. Импортирует его только core/transport.py;
# второго клиента («запасного») сознательно нет: это второй адаптер, второй путь
# ошибок и второй набор тестов ради поведения, которое вызывающий не видит.
httpx

# Разбор .env — один стандартный парсер: многострочные значения, кавычки,
# «export». Своим диалектом не занимаемся: `#` вне кавычек = комментарий,
# интерполяция `${VAR}` выключена (AGENTS §7).
python-dotenv

# Прогресс CLI-стадий (в web-режиме гасится WEB_PROGRESS=1)
tqdm

# Опционально: ускоряет поиск терминов (Aho-Corasick вместо regex-перебора)
pyahocorasick

# Не ставим осознанно:
#   requests — HTTP-клиент в проекте один (httpx);
#   tenacity — политики ретраев одна и она не про исключения (гигиена стрима);
#   psutil — снятие дерева процессов закрыто stdlib (killpg / taskkill), а пакет
#     добавляет бинарное колесо и третью ветку остановки;
#   uvicorn/fastapi/waitress — web-слой синхронный и однопользовательский,
#     ASGI/WSGI-переписывание не решает нашу задачу (SSE-соединений столько,
#     сколько открытых вкладок);
#   htmx / alpine — SPA на ванильном JS, вторая модель DOM-обновлений не нужна;
#   dayjs — даты форматирует Intl.DateTimeFormat браузера;
#   chart.js — графиков в интерфейсе нет (статистика таблицами);
#   diff-match-patch — сравнение глав сделано на LCS по абзацам;
#   tiktoken — тянет BPE-словари из сети, ломает офлайн-установку.
