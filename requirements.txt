# NovelMaestro: зависимости рантайма (pip install -r requirements.txt)
#
# Политика (AGENTS.md §2): внешних библиотек минимум; каждая опциональна —
# без неё работает встроенный фолбэк, поэтому установка без интернета и на
# экзотической платформе не падает. Активный стек печатает
# `python3 -m core.deps` (и он же в логе web-сервера при старте).
#
# Разработка (тесты) — requirements-dev.txt, он включает этот файл.

# HTTP-транспорт LLM. Роль одна, кандидата два: httpx — основной (конвейер
# переиспользует соединения вместо нового handshake на каждый запрос, таймауты
# connect/write/read разделены), requests — фолбэк, который есть почти во всех
# дистрибутивах. Импортирует их только core/transport.py.
httpx
requests

# Прогресс CLI-стадий (в web-режиме гасится WEB_PROGRESS=1)
tqdm

# Опционально: ускоряет поиск терминов (Aho-Corasick вместо regex-перебора)
pyahocorasick

# Не ставим осознанно:
#   python-dotenv — его парсер съедаёт «#» в значениях (в regexp-полях .env
#     «#» — литерал паттерна) и раскрывает ${VAR}; собственный парсер в
#     core/common.py делает ровно то, что нужно проекту;
#   tenacity — политики ретраев одна и она не про исключения (гигиена стрима);
#   psutil — снятие дерева процессов уже реализовано на stdlib (killpg /
#     taskkill), а пакет добавляет бинарное колесо и третью ветку остановки;
#   uvicorn/fastapi/waitress — web-слой синхронный и однопользовательский,
#     ASGI/WSGI-переписывание не решает нашу задачу (SSE-соединений столько,
#     сколько открытых вкладок);
#   htmx / alpine — SPA на ванильном JS, вторая модель DOM-обновлений не нужна;
#   dayjs — даты форматирует Intl.DateTimeFormat браузера;
#   chart.js — графиков в интерфейсе нет (статистика таблицами);
#   diff-match-patch — сравнение глав сделано на LCS по абзацам;
#   tiktoken — тянет BPE-словари из сети, ломает офлайн-установку.
