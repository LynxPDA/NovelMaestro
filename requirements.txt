# NovelMaestro: зависимости (pip install -r requirements.txt)
#
# Политика (AGENTS.md §2): внешних библиотек минимум; каждая опциональна —
# без неё работает встроенный фолбэк, поэтому установка без интернета и на
# экзотической платформе не падает. Активный стек печатает
# `python3 -m core.deps` (и он же в логе web-сервера при старте).

# Обязательные
requests        # HTTP-транспорт LLM (фолбэк; ставится почти везде)
tqdm            # прогресс CLI-стадий (в web-режиме гасится WEB_PROGRESS=1)

# Рекомендуемые: основной транспорт LLM. С httpx конвейер переиспользует
# соединения (пул) вместо нового handshake на каждый запрос и получает
# раздельные connect/write/read таймауты. Без него всё работает на requests.
httpx

# Опционально: ускоряет поиск терминов (Aho-Corasick вместо regex-перебора)
pyahocorasick

# Разработка (тесты)
pytest

# Не ставим осознанно:
#   python-dotenv — его парсер съедаёт «#» в значениях (в regexp-полях .env
#     «#» — литерал паттерна) и раскрывает ${VAR}; собственный парсер в
#     core/common.py делает ровно то, что нужно проекту;
#   tenacity — политики ретраев одна и она не про исключения (гигиена стрима);
#   psutil — снятие дерева процессов уже реализовано на stdlib (killpg /
#     taskkill), а пакет добавляет бинарное колесо и третью ветку остановки;
#   uvicorn/fastapi — web-слой синхронный и однопользовательский, ASGI-переписывание
#     не решает нашу задачу (SSE-соединений столько, сколько открытых вкладок);
#   tiktoken — тянет BPE-файлы из сети, ломает офлайн-установку.
